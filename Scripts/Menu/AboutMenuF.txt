event ObjectUpdate
	object.alpha = optionsMenuH.alpha
	if object.alpha > 254
		if 2 == 1	// This dummy-looking origins code was added to ignore inputs
			switch object.state
			case ABOUTMENU_CONTROLS
				if options.physicalControls == true
					if keyPress[0].up == true
						object.buttonSelected--
						if object.buttonSelected < 0
							object.buttonSelected = 2
						end if
						PlaySfx(SFX_G_MENUBUTTON, false)
					end if

					if keyPress[0].down == true
						object.buttonSelected++
						if object.buttonSelected > 2
							object.buttonSelected = 0
						end if
						PlaySfx(SFX_G_MENUBUTTON, false)
					end if

					CheckTouchRect(0, 0, screen.xsize, screen.ysize)	//Touch input reset
					if checkResult > -1
						options.physicalControls = false
						object.buttonSelected = -1
					end if
				else
					CheckTouchRect(0, 0, screen.xsize, screen.ysize)	//Touch input reset
					temp3 = checkResult

					temp0  = screen.xcenter
					temp0 += 36
					temp1  = screen.xcenter
					temp1 += 110
					CheckTouchRect(temp0, 108, temp1, 134)	//Terms button press
					if checkResult > -1
						object.buttonSelected = 0
					else
						if temp3 < 0
							if object.buttonSelected == 0
								keyPress[0].start = true
							end if
						else
							if object.buttonSelected == 0
								object.buttonSelected = -1
							end if
						end if
					end if

					temp0  = screen.xcenter
					temp0 += 36
					temp1  = screen.xcenter
					temp1 += 110
					CheckTouchRect(temp0, 134, temp1, 160)	//Privacy button press
					if checkResult > -1
						object.buttonSelected = 1
					else
						if temp3 < 0
							if object.buttonSelected == 1
								keyPress[0].start = true
							end if
						else
							if object.buttonSelected == 1
								object.buttonSelected = -1
							end if
						end if
					end if

					temp0  = screen.xcenter
					temp0 += 36
					temp1  = screen.xcenter
					temp1 += 110
					CheckTouchRect(temp0, 160, temp1, 185)	//Facebook button press
					if checkResult > -1
						object.buttonSelected = 2
					else
						if temp3 < 0
							if object.buttonSelected == 2
								keyPress[0].start = true
							end if
						else
							if object.buttonSelected == 2
								object.buttonSelected = -1
							end if
						end if
					end if

					CallFunction(MenuHeading_CheckTouchBack)
					if keyPress[0].up == true
						object.buttonSelected = 1
						options.physicalControls = true
					end if

					if keyPress[0].down == true
						object.buttonSelected = 0
						options.physicalControls = true
					end if

				end if

				checkResult = false
				if object.buttonSelected > -1
					if keyPress[0].start == true
						checkResult = true
					end if

					if keyPress[0].buttonA == true
						checkResult = true
					end if
				end if

				if checkResult == true
					PlaySfx(SFX_G_SELECT, false)
					object.state++
				end if
				break

			case ABOUTMENU_OPTION_SELECTED
				if object.animationTimer < 30
					object.animationTimer++
				else
					object.animationTimer = 0
					object.state = ABOUTMENU_CONTROLS
					switch object.buttonSelected
					case 0
						// engineCallback(CALLBACK_TERMS_SELECTED)
						break

					case 1
						// engineCallback(CALLBACK_PRIVACY_SELECTED)
						break

					case 2
						// engineCallback(CALLBACK_FACEBOOK_LOG)
						break

					end switch
					if options.physicalControls == false
						object.buttonSelected = -1
					end if
				end if
				break

			end switch
		end if
	else
		if options.physicalControls == false
			object.buttonSelected = -1
		end if

		object.animationTimer = 0
		object.state = ABOUTMENU_CONTROLS
	end if
end event

	temp2 <<= 2
	FlipSign(temp2)
	temp2  += screen.xcenter
	temp2  += 72

		default
			object.frame = 0
			if temp0 > 64
				if temp0 < 91
					object.frame  = temp0
					object.frame -= 63
				end if
			end if
			if temp0 > 96
				if temp0 < 123
					object.frame  = temp0
					object.frame -= 95
				end if
			end if
			if temp0 > 47
				if temp0 < 58

				end if
			end if
			if object.frame > 0
				DrawSpriteScreenFX(object.frame, FX_INK, temp2, 92)
			end if
			break

		case 38
