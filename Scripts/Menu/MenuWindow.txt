private alias object.value4 : object.yOffset

event ObjectUpdate
	switch object.state
	case MENUWINDOW_EXPAND
		if object.timer < 14
			object.timer++
			//Width
			tempValue0    = 300
			tempValue0   -= object.width
			tempValue0  >>= 3
			object.width += tempValue0
			//Height
			tempValue0     = 224
			tempValue0    -= object.height
			tempValue0   >>= 3
			object.height += tempValue0
			//XPos
			tempValue0   = screen.xcenter
			tempValue0 <<= 16
			tempValue0  -= object.xpos
			tempValue0 >>= 2
			object.xpos += tempValue0
			//YPos
			tempValue0   = screen.ycenter
			tempValue0  -= 6
			tempValue0 <<= 16
			tempValue0  -= object.ypos
			tempValue0 >>= 2
			object.ypos += tempValue0
		else
			object.timer = 0
			object.state++
		end if
		break
		
	case MENUWINDOW_SCALEDOWN
		if object.timer < 30
			object.timer++
			// Width
			tempValue0    = 256
			tempValue0   -= object.width
			tempValue0  >>= 2
			object.width += tempValue0
			// Height
			tempValue0     = 160
			tempValue0    -= object.height
			tempValue0   >>= 2
			object.height += tempValue0
			// XPos
			tempValue0   = screen.xcenter
			tempValue0 <<= 16
			if object.xpos > tempValue0
				object.xpos -= 0x10000
			end if

			if object.xpos < tempValue0
				object.xpos += 0x10000
			end if
			// YPos
			tempValue0   = screen.ycenter
			tempValue0  -= 8
			tempValue0 <<= 16
			if object.ypos > tempValue0
				object.ypos -= 0x10000
			end if

			if object.ypos < tempValue0
				object.ypos += 0x10000
			end if
		else
			object.timer = 0
			object.state++
		end if
		break

	case MENUWINDOW_BLANK
		// Nothing
		break

	case MENUWINDOW_SHRINK
		if object.timer < 14
			object.timer++
			// Width
			tempValue0   = 132
			tempValue0  -= object.width
			tempValue0 >>= 3
			object.width += tempValue0
			// Height
			tempValue0   = 18
			tempValue0  -= object.height
			tempValue0 >>= 3
			object.height += tempValue0
			// XPos
			tempValue0   = object.xOffset
			tempValue0  -= object.xpos
			tempValue0 >>= 2
			object.xpos += tempValue0
			// YPos
			tempValue0   = object.yOffset
			tempValue0  -= object.ypos
			tempValue0 >>= 2
			object.ypos += tempValue0
		else
			object.timer = 0
			object.state++
		end if
		break
		
	case MENUWINDOW_FIT_IN_BUTTON
		if object.timer < 14
			object.timer++
			// Width
			tempValue0   = 144
			tempValue0  -= object.width
			tempValue0 >>= 3
			object.width += tempValue0
			// Height
			tempValue0   = 21
			tempValue0   -= object.height
			tempValue0 >>= 3
			object.height += tempValue0
			// XPos
			if object.xpos > object.xOffset
				object.xpos -= 0x10000
			end if

			if object.xpos < object.xOffset
				object.xpos += 0x10000
			end if
			// YPos
			if object.ypos > object.yOffset
				object.ypos -= 0x10000
			end if

			if object.ypos < object.yOffset
				object.ypos += 0x10000
			end if
		else
			object.timer++
			if object.timer == 16
				object.timer = 0
				object.type = TypeName[Blank Object]
			end if

			arrayPos0 = object.entityPos
			object[arrayPos0].type = TypeName[Menu Button]
			object[arrayPos0].state = MENUBUTTON_RESET
		end if
		break

	case MENUWINDOW_SOUNDTRACK_EXPAND
		if object.timer < 14
			object.timer++
			// Width
			tempValue0    = 300
			tempValue0   -= object.width
			tempValue0  >>= 3
			object.width += tempValue0
			// Height
			tempValue0     = 192
			tempValue0    -= object.height
			tempValue0   >>= 3
			object.height += tempValue0
			// XPos
			tempValue0   = screen.xcenter
			tempValue0 <<= 16
			tempValue0  -= object.xpos
			tempValue0 >>= 2
			object.xpos += tempValue0
			// YPos
			tempValue0   = screen.ycenter
			tempValue0  -= 6
			tempValue0 <<= 16
			tempValue0  -= object.ypos
			tempValue0 >>= 2
			object.ypos += tempValue0
		else
			object.timer = 0
			object.state++
		end if
		break

	case MENUWINDOW_SOUNDTRACK_SHRINK
		if object.timer < 30
			object.timer++
			// Width
			tempValue0    = 256
			tempValue0   -= object.width
			tempValue0  >>= 2
			object.width += tempValue0
			// Height
			tempValue0     = 128
			tempValue0    -= object.height
			tempValue0   >>= 2
			object.height += tempValue0
			// XPos
			tempValue0   = screen.xcenter
			tempValue0 <<= 16
			if object.xpos > tempValue0
				object.xpos -= 0x10000
			end if
			if object.xpos < tempValue0
				object.xpos += 0x10000
			end if
			// YPos
			tempValue0   = screen.ycenter
			tempValue0  -= 8
			tempValue0 <<= 16
			if object.ypos > tempValue0
				object.ypos -= 0x10000
			end if
			if object.ypos < tempValue0
				object.ypos += 0x10000
			end if
		else
			object.timer = 0
			object.state = MENUWINDOW_BLANK
		end if
		break

	end switch
end event


event ObjectDraw
	//Upper Left Border
	tempValue0  = object.width
	tempValue0 *= -0x8000
	tempValue0 += object.xpos
	tempValue1  = object.height
	tempValue1 *= -0x8000
	tempValue1 += object.ypos
	DrawSpriteXY(0, tempValue0, tempValue1)

	//Upper Right Border
	tempValue0   = object.width
	tempValue0 <<= 15
	tempValue0  += object.xpos
	DrawSpriteXY(1, tempValue0, tempValue1)

	//Bottom Left Border
	tempValue0   = object.width
	tempValue0  *= -0x8000
	tempValue0  += object.xpos
	tempValue1   = object.height
	tempValue1 <<= 15
	tempValue1  += object.ypos
	DrawSpriteXY(2, tempValue0, tempValue1)

	//Bottom Right Border
	tempValue0   = object.width
	tempValue0 <<= 15
	tempValue0  += object.xpos
	DrawSpriteXY(3, tempValue0, tempValue1)

	//Menu Filling
	tempValue0   = object.width
	tempValue0  *= -0x8000
	tempValue0  += object.xpos
	tempValue0 >>= 16
	tempValue1   = object.height
	tempValue1  *= -0x8000
	tempValue1  += object.ypos
	tempValue1  += 0xA0000
	tempValue1 >>= 16
	tempValue2   = object.width
	tempValue2  -= 2
	tempValue3   = object.height
	tempValue3  -= 21
	DrawRect(tempValue0, tempValue1, tempValue2, tempValue3, 96, 96, 128, 255)

	//Menu Shadow
	tempValue0   = object.width
	tempValue0 <<= 15
	tempValue0  += object.xpos
	tempValue0  -= 0x20000
	tempValue0 >>= 16
	tempValue2   = 2
	DrawRect(tempValue0, tempValue1, tempValue2, tempValue3, 0, 0, 0, 255)

	object.inkEffect = INK_ALPHA
	switch object.state
	case MENUWINDOW_EXPAND
	case MENUWINDOW_SCALEDOWN
	case MENUWINDOW_SOUNDTRACK_EXPAND
	case MENUWINDOW_SOUNDTRACK_SHRINK
		if object.alpha < 248
			object.alpha += 5
		end if
		break

	case MENUWINDOW_BLANK
		object.alpha = 255
		break

	case MENUWINDOW_SHRINK
	case MENUWINDOW_FIT_IN_BUTTON
		if object.alpha > 0
			object.alpha -= 8
		end if
		break
		
	end switch
	tempValue0  = screen.xsize
	tempValue0 -= 48
	if object.backTouchPress == true
		DrawSpriteScreenFX(5, FX_INK, tempValue0, 225)
	else
		DrawSpriteScreenFX(4, FX_INK, tempValue0, 225)
	end if
end event

		LoadSpriteSheet("Menu/MenuGfx2_FR.gif")
		break
	case LANG_ITALIAN

		break
	case LANG_DEUTSCH

		break

		break

	// (#0 - #3) - Borders for the Menu Window, Upper Left - Right - Bottom Left - Bottom Right
