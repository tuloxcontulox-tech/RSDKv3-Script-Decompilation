reserve function OptionsMenuH_DrawOptions

private function OptionsMenuH_DrawOptions
	switch object.flickerText
	case false
		if arrayPos0 == object.currentSelection
			DrawSpriteScreenFX(9, FX_INK, object.xpos, object.ypos)
			temp0  = arrayPos0
			temp0 += 14
			DrawSpriteScreenFX(temp0, FX_INK, object.xpos, object.ypos)
		else
			temp0  = arrayPos0
			temp0 += 10
			DrawSpriteScreenFX(temp0, FX_INK, object.xpos, object.ypos)
		end if
		break

	case true
		temp0   = object.timer
		temp0  &= 3
		temp0 >>= 1
		DrawSpriteScreenFX(9, FX_INK, object.xpos, object.ypos)
		if temp0 == 0
			temp0  = arrayPos0
			temp0 += 14
			DrawSpriteScreenFX(temp0, FX_INK, object.xpos, object.ypos)
		end if
		break

	end switch
	
	DrawSpriteScreenFX(8, FX_INK, object.xpos, object.ypos)
	DrawSpriteScreenFX(arrayPos0, FX_INK, object.xpos, object.ypos)
end function


event ObjectUpdate
	switch object.state
	case OPTIONS_SETUP
		if options.physicalControls == false
			object.currentSelection = -1
		end if

		object.xpos  = screen.xcenter
		object.xpos -= 126

		if object.timer < 256
			object.alpha  = object.timer
			object.timer += 6
		else
			object.timer = 0
			object.alpha = 255
			object.state++
		end if
		break

	case OPTIONS_CONTROLS

		if options.physicalControls == true
			if keyPress[0].up == true
				object.currentSelection--
				if object.currentSelection < 0
					object.currentSelection = 3
				end if
				PlaySfx(SFX_G_MENUBUTTON, false)
			end if

			if keyPress[0].down == true
				object.currentSelection++
				if object.currentSelection > 3
					object.currentSelection = 0
				end if
				PlaySfx(SFX_G_MENUBUTTON, false)
			end if

			CheckTouchRect(0, 0, screen.xsize, screen.ysize)
			if checkResult > -1
				options.physicalControls = false
				object.currentSelection = -1
			end if
		else
			CheckTouchRect(0, 0, screen.xsize, screen.ysize)
			temp3 = checkResult

			temp0 = 0
			CallFunction(LoadSaveMenu_CheckTouchSave)
			if checkResult > -1
				object.currentSelection = 0
			else
				if temp3 < 0
					if object.currentSelection == 0
						keyPress[0].start = true
					end if
				else
					if object.currentSelection == 0
						object.currentSelection = -1
					end if
				end if
			end if

			temp0 = 1
			CallFunction(LoadSaveMenu_CheckTouchSave)
			if checkResult > -1
				object.currentSelection = 1
			else
				if temp3 < 0
					if object.currentSelection == 1
						keyPress[0].start = true
					end if
				else
					if object.currentSelection == 1
						object.currentSelection = -1
					end if
				end if
			end if

			temp0 = 2
			CallFunction(LoadSaveMenu_CheckTouchSave)
			if checkResult > -1
				object.currentSelection = 2
			else
				if temp3 < 0
					if object.currentSelection == 2
						keyPress[0].start = true
					end if
				else
					if object.currentSelection == 2
						object.currentSelection = -1
					end if
				end if
			end if

			temp0 = 3
			CallFunction(LoadSaveMenu_CheckTouchSave)
			if checkResult > -1
				object.currentSelection = 3
			else
				if temp3 < 0
					if object.currentSelection == 3
						keyPress[0].start = true
					end if
				else
					if object.currentSelection == 3
						object.currentSelection = -1
					end if
				end if
			end if
			CallFunction(MenuHeading_CheckTouchBack)
			
			if keyPress[0].up == true
				object.currentSelection  = 3
				options.physicalControls = true
			end if

			if keyPress[0].down == true
				object.currentSelection  = 0
				options.physicalControls = true
			end if
		end if

		checkResult = false
		if object.currentSelection > -1
			if keyPress[0].start == true
				checkResult = true
			end if
			if keyPress[0].buttonA == true
				checkResult = true
			end if
		end if

		if checkResult == true
			if object.currentSelection < 4
				arrayPos0 = object.currentSelection
				object.flickerText = 1
				object.state++
				switch object.currentSelection
				case 0
					StopMusic()
					break
				case 1
					break
				case 2
					StopMusic()
					break
				case 3
					break
				end switch
				PlaySfx(SFX_G_SELECT, false)
			end if
		end if

		if keyPress[0].buttonB == true
			object.timer = 248
			object.state = OPTIONS_EXIT
			PlaySfx(SFX_S_BACKBUTTON, false)
		end if
		break

	case OPTIONS_SELECTED
		if object.timer < 30
			object.timer++
		else
			object.timer = 0
			arrayPos0 = object.currentSelection
			object.flickerText = 0
			switch object.currentSelection
			case 0
				object.state = OPTIONS_STAGE_LOAD
				break

			case 1
				ReadSaveRAM()
				if SaveRAM[32] == false
					SaveRAM[33] = 100
					SaveRAM[34] = 100
					SaveRAM[35] = 0
					SaveRAM[40] = true	// Haptics
				end if

				object.state = OPTIONS_SETTINGS_SETUP

				object.ypos  = 40
				object.ypos *= object.currentSelection
				object.ypos += screen.ycenter
				object.ypos -= 87
				break

			case 2
				object.state = OPTIONS_STAGE_LOAD
				break

			case 3
				object.state = OPTIONS_ABOUTMENU_SETUP
				object.ypos  = 40
				object.ypos *= object.currentSelection
				object.ypos += screen.ycenter
				object.ypos -= 87
				break
			end switch
		end if
		break

	case OPTIONS_STAGE_LOAD
		if object.timer < 320
			object.timer += 8
		else
			switch object.currentSelection
			case 0
				if options.devMenuFlag == true
					engine.state = RESET_GAME
				else
#platform: Use_Standalone
					Stage.listPos = STAGE_P_HELP
#endplatform
#platform: Use_Origins
					if engine.deviceType == RETRO_MOBILE
						Stage.listPos = STAGE_P_HELP
					else
						// EngineCallback(CALLBACK_RETURNSTORE_SELECTED)
					end if
#endplatform
					LoadStage()
				end if
				break

			case 2
				Stage.listPos = STAGE_P_STAFFCREDITS
				LoadStage()
				break

			case 3 // Unused, seems to just be here to summon the Dev Menu though
				engine.state = RESET_GAME
				break
			end switch
		end if

		temp0 = object.timer
		if temp0 > 255
			temp0 = 255
		end if
		SetScreenFade(0, 0, 0, temp0)
		break

	case OPTIONS_SETTINGS_SETUP
		temp0  = screen.ycenter
		temp0 -= 87
		if object.ypos > temp0
			object.ypos -= 8
			if object.ypos < temp0
				object.ypos = temp0
			end if
		else
			object.state++
			object.alpha = 0
			object.optionCount = 0
			if options.physicalControls == false
				arrayPos0 = object.currentSelection
				object.optionCount = -1
			end if
		end if
		break

	case OPTIONS_SETTINGS_POPIN
		if object.timer < 256
			object.alpha  = object.timer
			object.timer += 8
		else
			object.timer = 0
			object.alpha = 255
			object.state++
		end if
		break
	case OPTIONS_SETTINGS_CONTROLS
		arrayPos0 = object.currentSelection
		if options.physicalControls == true
			if keyPress[0].up == true
				object.optionCount--
				PlaySfx(SFX_G_MENUBUTTON, false)
				if object.optionCount < 0
					object.optionCount = 3
				end if
			end if
			if keyPress[0].down == true
				object.optionCount++
				PlaySfx(SFX_G_MENUBUTTON, false)
				if object.optionCount > 3
					object.optionCount = 0
				end if
			end if
			switch object.optionCount
			case 0	// BGM
				if keyPress[0].left == true
					object.bGMDecreaseTimer = 10
					SaveRAM[33] -= 10
					if SaveRAM[33] < 0
						SaveRAM[33] = 0
						options.bgmVolume = SaveRAM[33]
					else
						options.bgmVolume = SaveRAM[33]
						PlaySfx(SFX_G_MENUBUTTON, false)
					end if
				end if

				if keyPress[0].right == true
					object.bGMIncreaseTimer = 10
					SaveRAM[33] += 10
					if SaveRAM[33] > 100
						SaveRAM[33] = 100
						options.bgmVolume = SaveRAM[33]
					else
						options.bgmVolume = SaveRAM[33]
						PlaySfx(SFX_G_MENUBUTTON, false)
					end if
				end if
				break

			case 1	// SFX
				if keyPress[0].left == true
					object.sFXDecreaseTimer = 10
					SaveRAM[34] -= 10
					if SaveRAM[34] < 0
						SaveRAM[34] = 0
						options.sfxVolume = SaveRAM[34]
					else
						PlaySfx(SFX_G_MENUBUTTON, false)
						options.sfxVolume = SaveRAM[34]
					end if
				end if

				if keyPress[0].right == true
					object.sFXIncreaseTimer = 10
					SaveRAM[34] += 10
					if SaveRAM[34] > 100
						SaveRAM[34] = 100
					else
						PlaySfx(SFX_G_MENUBUTTON, false)
						options.sfxVolume = SaveRAM[34]
					end if
				end if
				break

			case 2	// Control Type
				if keyPress[0].left == true
					SaveRAM[35]--
					if SaveRAM[35] < 0
						SaveRAM[35] = 1
					end if
					PlaySfx(SFX_G_MENUBUTTON, false)
				end if

				if keyPress[0].right == true
					SaveRAM[35]++
					if SaveRAM[35] > 1
						SaveRAM[35] = 0
					end if
					PlaySfx(SFX_G_MENUBUTTON, false)
				end if
				break

			case 3	// Haptics
				if keyPress[0].left == true
					SaveRAM[40]--
					if SaveRAM[40] < false
						SaveRAM[40] = true
						// HapticEffect(30, 0, 0, 0)
					end if
					PlaySfx(SFX_G_MENUBUTTON, false)
				end if

				if keyPress[0].right == true
					SaveRAM[40]++
					if SaveRAM[40] > true
						SaveRAM[40] = false
					end if
					PlaySfx(SFX_G_MENUBUTTON, false)
				end if
				break
			end switch

			CheckTouchRect(0, 0, screen.xsize, screen.ysize)
			if checkResult > -1
				options.physicalControls = false
				object.optionCount = -1
			end if
		else
			CheckTouchRect(0, 0, screen.xsize, screen.ysize)
			if checkResult > -1
				object.touchDelay++
				object.touchDelay %= 24
			else
				object.touchDelay = 0
			end if

			temp3  = checkResult
			temp0  = screen.xcenter
			temp0 += 16
			temp1  = screen.xcenter
			temp1 += 48
			CheckTouchRect(temp0, 70, temp1, 97)
			if checkResult > -1
				object.optionCount		= 3
				object.bGMDecreaseTimer = 10
				if object.touchDelay == 23
					SaveRAM[33] -= 10
					if SaveRAM[33] < 0
						SaveRAM[33] = 0
						options.bgmVolume = SaveRAM[33]
					else
						options.bgmVolume = SaveRAM[33]
						PlaySfx(SFX_G_MENUBUTTON, false)
					end if
				end if
			else
				if temp3 < 0
					if object.optionCount == 3
						object.bGMDecreaseTimer = 0
						SaveRAM[33] -= 10
						if SaveRAM[33] < 0
							SaveRAM[33] = 0
							options.bgmVolume = SaveRAM[33]
						else
							options.bgmVolume = SaveRAM[33]
							PlaySfx(SFX_G_MENUBUTTON, false)
						end if
						object.optionCount = -1
					end if
				else
					if object.optionCount == 3
						object.optionCount = -1
					end if
				end if
			end if
			CheckTouchRect(temp0, 99, temp1, 126)
			if checkResult > -1
				object.optionCount		= 4
				object.sFXDecreaseTimer = 10
				if object.touchDelay == 23
					SaveRAM[34] -= 10
					if SaveRAM[34] < 0
						SaveRAM[34] = 0
						options.sfxVolume = SaveRAM[34]
					else
						options.sfxVolume = SaveRAM[34]
						PlaySfx(SFX_G_MENUBUTTON, false)
					end if
				end if
			else
				if temp3 < 0
					if object.optionCount == 4
						object.sFXDecreaseTimer = 0
						SaveRAM[34] -= 10
						if SaveRAM[34] < 0
							SaveRAM[34] = 0
							options.sfxVolume = SaveRAM[34]
						else
							options.sfxVolume = SaveRAM[34]
							PlaySfx(SFX_G_MENUBUTTON, false)
						end if
						object.optionCount = -1
					end if
				else
					if object.optionCount == 4
						object.optionCount = -1
					end if
				end if
			end if

			temp0  = screen.xcenter
			temp0 += 84
			temp1  = screen.xcenter
			temp1 += 120
			CheckTouchRect(temp0, 70, temp1, 97)
			if checkResult > -1
				object.optionCount		= 5
				object.bGMIncreaseTimer = 10
				if object.touchDelay == 23
					SaveRAM[33] += 10
					if SaveRAM[33] > 100
						SaveRAM[33] = 100
						options.bgmVolume = SaveRAM[33]
					else
						options.bgmVolume = SaveRAM[33]
						PlaySfx(SFX_G_MENUBUTTON, false)
					end if
				end if
			else
				if temp3 < 0
					if object.optionCount == 5
						object.bGMIncreaseTimer = 0
						SaveRAM[33] += 10
						if SaveRAM[33] > 100
							SaveRAM[33] = 100
							options.bgmVolume = SaveRAM[33]
						else
							options.bgmVolume = SaveRAM[33]
							PlaySfx(SFX_G_MENUBUTTON, false)
						end if
						object.optionCount = -1
					end if
				else
					if object.optionCount == 5
						object.optionCount = -1
					end if
				end if
			end if

			CheckTouchRect(temp0, 99, temp1, 126)
			if checkResult > -1
				object.optionCount		= 6
				object.sFXIncreaseTimer = 10
				if object.touchDelay == 23
					SaveRAM[34] += 10
					if SaveRAM[34] > 100
						SaveRAM[34] = 100
						options.sfxVolume = SaveRAM[34]
					else
						options.sfxVolume = SaveRAM[34]
						PlaySfx(SFX_G_MENUBUTTON, false)
					end if
				end if
			else
				if temp3 < 0
					if object.optionCount == 6
						object.sFXIncreaseTimer = 0
						SaveRAM[34] += 10
						if SaveRAM[34] > 100
							SaveRAM[34] = 100
							options.sfxVolume = SaveRAM[34]
						else
							options.sfxVolume = SaveRAM[34]
							PlaySfx(SFX_G_MENUBUTTON, false)
						end if
						object.optionCount = -1
					end if
				else
					if object.optionCount == 6
						object.optionCount = -1
					end if
				end if
			end if

			temp0  = screen.xcenter
			temp0 += 14
			temp1  = screen.xcenter
			temp1 += 64
			CheckTouchRect(temp0, 130, temp1, 158)
			if checkResult > -1
				object.optionCount = 7
			else
				if temp3 < 0
					if object.optionCount == 7
						SaveRAM[35] = 0
						PlaySfx(SFX_G_MENUBUTTON, false)
						object.optionCount = -1
					end if
				else
					if object.optionCount == 7
						object.optionCount = -1
					end if
				end if
			end if

			temp0  = screen.xcenter
			temp0 += 68
			temp1  = screen.xcenter
			temp1 += 114
			CheckTouchRect(temp0, 130, temp1, 158)
			if checkResult > -1
				object.optionCount = 8
			else
				if temp3 < 0
					if object.optionCount == 8
						SaveRAM[35] = 1
						PlaySfx(SFX_G_MENUBUTTON, false)
						object.optionCount = -1
					end if
				else
					if object.optionCount == 8
						object.optionCount = -1
					end if
				end if
			end if

			temp0  = screen.xcenter
			temp0 += 14
			temp1  = screen.xcenter
			temp1 += 64
			CheckTouchRect(temp0, 160, temp1, 188)
			if checkResult > -1
				object.optionCount = 9
			else
				if temp3 < 0
					if object.optionCount == 9
						SaveRAM[40] = true
						options.hapticsEnabled = true
						PlaySfx(SFX_G_MENUBUTTON, false)
						object.optionCount = -1
						// HapticEffect(30, 0, 0, 0)
					end if
				else
					if object.optionCount == 9
						object.optionCount = -1
					end if
				end if
			end if

			temp0  = screen.xcenter
			temp0 += 68
			temp1  = screen.xcenter
			temp1 += 114
			CheckTouchRect(temp0, 160, temp1, 188)
			if checkResult > -1
				object.optionCount = 10
			else
				if temp3 < 0
					if object.optionCount == 10
						SaveRAM[40] = false
						options.hapticsEnabled = false
						PlaySfx(SFX_G_MENUBUTTON, false)
						object.optionCount = -1
					end if
				else
					if object.optionCount == 10
						object.optionCount = -1
					end if
				end if
			end if

			temp0  = screen.xsize
			temp0 -= 96
			CheckTouchRect(temp0, 192, screen.xsize, screen.ysize)
			if checkResult > -1
				object.optionCount = 11
				object.backTouchPress = true
			else
				object.backTouchPress = false
				if temp3 < 0
					if object.optionCount == 11
						keyPress[0].buttonB = 1
					end if
				else
					if object.optionCount == 11
						object.optionCount = -1
					end if
				end if
			end if

			if keyPress[0].up == true
				object.optionCount = 2
				options.physicalControls = true
			end if
			
			if keyPress[0].down == true
				object.optionCount = 0
				options.physicalControls = true
			end if
		end if

		if keyPress[0].buttonB == true
			object.state = OPTIONS_EXIT_SETTINGS
			object.timer = 248
			options.originalControls = SaveRAM[35]
			options.hapticsEnabled	 = SaveRAM[40]
			SaveRAM[32] = true
			WriteSaveRAM()
			PlaySfx(SFX_S_BACKBUTTON, false)
		end if
		break

	case OPTIONS_UNUSED
		if object.timer < 30
			object.timer++
		else
			object.timer = 0
			object.state++
		end if
		break

	case OPTIONS_UNUSED2
		if object.timer > 0
			object.timer -= 8
		else
			if options.physicalControls == false
				object.currentSelection = -1
			end if
			object.state = OPTIONS_CONTROLS
			PlayMusic(0)
		end if

		temp0 = object.timer
		if temp0 > 255
			temp0 = 255
		end if
		SetScreenFade(0, 0, 0, temp0)
		break

	case OPTIONS_EXIT_SETTINGS
		if object.timer > 0
			object.alpha  = object.timer
			object.timer -= 8
		else
			object.alpha = 0
			object.state++
		end if
		break

	case OPTIONS_LEAVE_SUBMENU
		temp0  = object.currentSelection
		temp0 *= 40
		temp0 += screen.ycenter
		temp0 -= 87

		if object.ypos < temp0
			object.ypos += 8
			if object.ypos > temp0
				object.ypos = temp0
			end if
			object.alpha = 0
		else
			if options.physicalControls == false
				object.currentSelection = -1
			end if
			object.state = OPTIONS_CONTROLS
			object.alpha = 255
		end if
		break

	case OPTIONS_EXIT
		if object.timer > 0
			object.alpha  = object.timer
			object.timer -= 8
		else
			object.type = TypeName[Blank Object]
			menuWindow.state = MENUWINDOW_SHRINK
			//All of this is MenuButton objects
			object[34].state = MENUBUTTON_BUTTONS_IN
			object[35].state = MENUBUTTON_BUTTONS_IN
			object[36].state = MENUBUTTON_BUTTONS_IN
			object[37].state = MENUBUTTON_BUTTONS_IN
			object[38].state = MENUBUTTON_BUTTONS_IN
			object[39].state = MENUBUTTON_BUTTONS_IN
			object[40].state = MENUBUTTON_BUTTONS_IN
			object[41].state = MENUBUTTON_BUTTONS_IN
			object[42].state = MENUBUTTON_BUTTONS_IN
			//End
			menuHeading.state  = MENUHEADING_LEAVE_SCREEN
			menuHeading.header = MAINMENU_HEADER
		end if
		break

	case OPTIONS_ABOUTMENU_SETUP
		temp0  = screen.ycenter
		temp0 -= 87
		if object.ypos > temp0
			object.ypos -= 8
			if object.ypos < temp0
				object.ypos = temp0
			end if
		else
			temp0  = object.entityPos
			temp0 += 10
			ResetObjectEntity(temp0, TypeName[About Menu F], 0, 0, 0)
			SetupMenu(MENU_1, 0, 3, 1)
			// GetVersionNumber(MENU_1, 1)
			object[+10].inkEffect = INK_ALPHA
			object.state++
			object.alpha = 0
			object.optionCount = 0
			if options.physicalControls == false
				arrayPos0 = object.currentSelection
				object.optionCount = -1
			end if
		end if
		break

	case OPTIONS_ENTER_ABOUTMENU
		if object.timer < 256
			object.alpha  = object.timer
			object.timer += 8
		else
			object.timer = 0
			object.alpha = 255
			object.state++
		end if
		break

	case OPTIONS_ABOUTMENU_CONTROLS
#platform: Mobile
		CheckTouchRect(0, 0, screen.xsize, screen.ysize)
		temp3  = checkResult
		temp0  = screen.xsize
		temp0 -= 96
		CheckTouchRect(temp0, 192, screen.xsize, screen.ysize)
		if checkResult > -1
			object.optionCount = 9
			object.backTouchPress = true
		else
			object.backTouchPress = false
			if temp3 < 0
				if object.optionCount == 9
					keyPress[0].buttonB = true
				end if
			else
				if object.optionCount == 9
					object.optionCount = -1
				end if
			end if
		end if
#endplatform

		if keyPress[0].buttonB == true
			object.state = OPTIONS_EXIT_ABOUTMENU
			object.timer = 248
			PlaySfx(SFX_S_BACKBUTTON, false)
		end if
		break

	case OPTIONS_EXIT_ABOUTMENU
		if object.timer > 0
			object.alpha  = object.timer
			object.timer -= 8
		else
			object.alpha = 0
			object.state = OPTIONS_LEAVE_SUBMENU
			object[+10].type = TypeName[Blank Object]
		end if
		break

	end switch
end event
